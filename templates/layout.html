errBox.classList.remove('hidden'); return; } localUsers.push(payload); localStorage.setItem('gm_registered_users', JSON.stringify(localUsers)); alert('Registrasi berhasil!'); switchToLoginModal(); document.getElementById('loginIdentifier').value = username; document.getElementById('loginPhone').value = phone; } // Login dengan Fallback Otomatis async function handleBuyerLoginSubmit(e) { e.preventDefault(); const errBox = document.getElementById('loginErrorMsg');
